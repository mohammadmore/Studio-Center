        id = parseInt(id);
        if (isChecked) {
            if (!window.selectedBatchIds.includes(id)) window.selectedBatchIds.push(id);

            window.selectedBatchIds = window.selectedBatchIds.filter(x => x !== id);
        }
        updateBatchUI();
    }
